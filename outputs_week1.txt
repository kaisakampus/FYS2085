   1.66666663

integer,parameter :: rk=selected_real_kind(r=200)
integer :: a=2, b=3
real :: x, y

! integers become real numbers when x and y is used: x=2.0, y=3.0
x=a
y=b

! integer division is rounded
(2+3)/3=5/3=1
! real number division is not rounded
(2.0+3.0)/3.0=1.66666663
! when the equation has at least one real number, then it is not rounded

! using larger real kind allows higher precision (more digits)
